"""
EDEN Optimization Mixins and Registry System
===========================================
Modular optimization system that allows combining multiple optimizations
without code duplication. Each optimization is implemented as a mixin
that can be composed with physics engines and other systems.

Key Features:
- Modular mixin architecture
- Centralized optimization registry  
- Automatic optimization combination
- Performance monitoring integration
- Configuration-driven optimization selection
"""

import asyncio
import logging
import time
from abc import ABC, abstractmethod
from dataclasses import dataclass, field
from typing import Any, Dict, List, Optional, Union, Callable, Type
from enum import Enum

try:
    import numpy as np
except ImportError:
    np = None

logger = logging.getLogger(__name__)


class OptimizationType(Enum):
    """Types of available optimizations."""
    PINGPONG = "pingpong"
    ETHICAL = "ethical"
    META_LEARNING = "meta_learning"
    HOMEOSTASIS = "homeostasis"
    MONITORING = "monitoring"
    JAX_ACCELERATION = "jax_acceleration"
    GPU_ACCELERATION = "gpu_acceleration"


@dataclass
class OptimizationResult:
    """Result of applying an optimization."""
    optimization_type: OptimizationType
    success: bool
    metrics: Dict[str, Any] = field(default_factory=dict)
    performance_gain: float = 0.0
    error_message: Optional[str] = None
    execution_time: float = 0.0


@dataclass
class OptimizationConfig:
    """Configuration for optimization applications."""
    enabled_optimizations: List[OptimizationType] = field(default_factory=list)
    optimization_intensity: float = 1.0
    auto_tune: bool = True
    performance_threshold: float = 0.05  # Minimum performance gain to apply
    error_tolerance: int = 3  # Max errors before disabling optimization


class OptimizationMixin(ABC):
    """Base mixin class for all optimizations."""
    
    def __init__(self, *args, **kwargs):
        super().__init__(*args, **kwargs)
        self._optimization_config = OptimizationConfig()
        self._optimization_metrics = {}
        self._optimization_errors = {}
    
    @property
    @abstractmethod
    def optimization_type(self) -> OptimizationType:
        """Return the type of this optimization."""
        pass
    
    @abstractmethod
    def apply_optimization(self, target_obj: Any, data: Any, entity_context: Dict[str, Any], 
                          system_metrics: Dict[str, Any], **kwargs) -> OptimizationResult:
        """
        Apply this optimization to the given data with access to the target object.
        
        Args:
            target_obj: The object being decorated/optimized (provides access to its state/methods)
            data: The data to optimize
            entity_context: Context about the entity being optimized (type, method, etc.)
            system_metrics: Current system performance metrics
            **kwargs: Additional optimization parameters
        
        Returns:
            OptimizationResult with success status and performance metrics
        """
        pass
    
    @abstractmethod
    def get_optimization_metrics(self) -> Dict[str, Any]:
        """Get current optimization metrics."""
        pass
    
    def configure_optimization(self, config: OptimizationConfig) -> None:
        """Configure this optimization."""
        self._optimization_config = config
    
    def is_optimization_enabled(self) -> bool:
        """Check if this optimization is enabled."""
        # Enable by default if no specific configuration
        if not hasattr(self._optimization_config, 'enabled_optimizations'):
            return True
        
        return (
            self.optimization_type in self._optimization_config.enabled_optimizations
            and self._get_error_count() < self._optimization_config.error_tolerance
        )
    
    def _get_error_count(self) -> int:
        """Get current error count for this optimization."""
        return self._optimization_errors.get(self.optimization_type.value, 0)
    
    def _record_error(self, error: Exception) -> None:
        """Record an optimization error."""
        error_key = self.optimization_type.value
        self._optimization_errors[error_key] = self._optimization_errors.get(error_key, 0) + 1
        logger.warning(f"Optimization {self.optimization_type.value} error: {error}")


class PingPongMixin(OptimizationMixin):
    """Mixin for ping-pong GPU buffer optimization with explicit JAX dependency injection."""
    
    def __init__(self, accelerator: Optional[Any] = None, *args, **kwargs):
        """
        Initialize PingPongMixin with optional JAX accelerator.
        
        Args:
            accelerator: Optional JAX accelerator strategy instance
            *args, **kwargs: Additional initialization parameters
        """
        super().__init__(*args, **kwargs)
        self._pingpong_engine = None
        self._pingpong_metrics = {}
        
        # Store accelerator (explicit dependency injection)
        self.accelerator = accelerator
        
        # Log accelerator type for transparency
        accelerator_type = type(accelerator).__name__ if accelerator else 'None'
        logger.info(f"PingPongMixin initialized with accelerator: {accelerator_type}")
    
    @property
    def optimization_type(self) -> OptimizationType:
        return OptimizationType.PINGPONG
    
    def apply_optimization(self, target_obj: Any, data: Any, entity_context: Dict[str, Any], 
                          system_metrics: Dict[str, Any], **kwargs) -> OptimizationResult:
        """Apply ping-pong optimization to data with access to target object."""
        start_time = time.time()
        
        try:
            if not self.is_optimization_enabled():
                return OptimizationResult(
                    optimization_type=self.optimization_type,
                    success=False,
                    error_message="Ping-pong optimization disabled"
                )
            
            # Initialize ping-pong engine if needed
            if self._pingpong_engine is None:
                self._initialize_pingpong()
            
            # Apply ping-pong optimization with accelerator delegation
            if self.accelerator and hasattr(self.accelerator, 'process_batch'):
                # Use JAX accelerator for high-performance processing
                logger.debug("Processing batch with JAX accelerator")
                optimized_data = self.accelerator.process_batch(data)
                performance_gain = self._calculate_performance_gain(data, optimized_data, start_time)
            elif self._pingpong_engine and hasattr(self._pingpong_engine, 'process_batch'):
                # Fallback to default ping-pong engine
                logger.debug("Processing batch with default ping-pong engine")
                if np is not None and isinstance(data, np.ndarray):
                    optimized_data = self._pingpong_engine.process_batch(data)
                    performance_gain = self._calculate_performance_gain(data, optimized_data, start_time)
                else:
                    # Convert data to numpy for processing
                    data_array = self._convert_to_array(data)
                    optimized_data = self._pingpong_engine.process_batch(data_array)
                    performance_gain = self._calculate_performance_gain(data_array, optimized_data, start_time)
            else:
                # Final fallback: no optimization available
                logger.debug("Processing batch with default non-accelerated implementation")
                optimized_data = self._process_batch_fallback(data)
                performance_gain = 0.0
            
            execution_time = time.time() - start_time
            
            # Update metrics
            self._pingpong_metrics.update({
                'last_execution_time': execution_time,
                'performance_gain': performance_gain,
                'successful_optimizations': self._pingpong_metrics.get('successful_optimizations', 0) + 1
            })
            
            return OptimizationResult(
                optimization_type=self.optimization_type,
                success=True,
                metrics=self._pingpong_metrics.copy(),
                performance_gain=performance_gain,
                execution_time=execution_time
            )
            
        except Exception as e:
            self._record_error(e)
            return OptimizationResult(
                optimization_type=self.optimization_type,
                success=False,
                error_message=str(e),
                execution_time=time.time() - start_time
            )
    
    def get_optimization_metrics(self) -> Dict[str, Any]:
        """Get ping-pong optimization metrics."""
        return self._pingpong_metrics.copy()
    
    def _initialize_pingpong(self) -> None:
        """Initialize ping-pong engine."""
        try:
            from crisalida_lib.EDEN.engines.pingpong_gpu_engine import create_pingpong_gpu_engine
            self._pingpong_engine = create_pingpong_gpu_engine()
        except ImportError:
            logger.warning("Ping-pong GPU engine not available")
            self._pingpong_engine = None
    
    def _convert_to_array(self, data: Any) -> Any:
        """Convert data to numpy array for processing."""
        if np is None:
            raise ImportError("NumPy not available for array conversion")
        
        if isinstance(data, (list, tuple)):
            return np.array(data, dtype=np.float32)
        elif isinstance(data, dict):
            # Convert dict to array of values
            return np.array(list(data.values()), dtype=np.float32)
        else:
            # Try to convert to float and create single-element array
            try:
                return np.array([float(data)], dtype=np.float32)
            except (ValueError, TypeError):
                # Fallback: create small random array
                return np.random.random((4,)).astype(np.float32)
    
    def _calculate_performance_gain(self, original: Any, optimized: Any, start_time: float) -> float:
        """Calculate performance gain from optimization."""
        # Simple heuristic: assume optimization provides gain based on execution time
        execution_time = time.time() - start_time
        if execution_time < 0.001:  # Very fast execution suggests good optimization
            return 0.2
        elif execution_time < 0.01:
            return 0.1
        else:
            return 0.05
    
    def _process_batch_fallback(self, data: Any) -> Any:
        """
        Fallback processing when no accelerator or engine is available.
        
        Args:
            data: Data to process
            
        Returns:
            Processed data (simple pass-through for fallback)
        """
        logger.debug("Using fallback processing (no optimization)")
        # Simple identity transformation as fallback
        return data


class EthicalOptimizationMixin(OptimizationMixin):
    """Mixin for ethical optimization based on entity wellbeing."""
    
    def __init__(self, *args, **kwargs):
        super().__init__(*args, **kwargs)
        self._ethical_engine = None
        self._ethical_metrics = {}
    
    @property
    def optimization_type(self) -> OptimizationType:
        return OptimizationType.ETHICAL
    
    def apply_optimization(self, target_obj: Any, data: Any, entity_context: Dict[str, Any], 
                          system_metrics: Dict[str, Any], **kwargs) -> OptimizationResult:
        """Apply ethical optimization to data with access to target object."""
        start_time = time.time()
        
        try:
            if not self.is_optimization_enabled():
                return OptimizationResult(
                    optimization_type=self.optimization_type,
                    success=False,
                    error_message="Ethical optimization disabled"
                )
            
            # Use provided entity_context for ethical evaluation, plus target_obj inspection
            wellbeing_scores = kwargs.get('wellbeing_scores', {})
            
            # Can now inspect target object for ethical considerations
            target_type = type(target_obj).__name__
            ethical_context = {**entity_context, 'target_type': target_type}
            
            # Apply ethical filtering and optimization
            ethical_score = self._evaluate_ethical_impact(data, ethical_context, wellbeing_scores)
            
            # Only proceed if ethically acceptable
            if ethical_score < 0.3:  # Below ethical threshold
                return OptimizationResult(
                    optimization_type=self.optimization_type,
                    success=False,
                    error_message="Operation failed ethical evaluation"
                )
            
            # Apply ethical optimization
            optimized_data = self._apply_ethical_constraints(data, ethical_score)
            performance_gain = ethical_score * 0.1  # Ethical operations get small performance bonus
            
            execution_time = time.time() - start_time
            
            # Update metrics
            self._ethical_metrics.update({
                'last_ethical_score': ethical_score,
                'ethical_optimizations': self._ethical_metrics.get('ethical_optimizations', 0) + 1,
                'avg_ethical_score': (
                    self._ethical_metrics.get('avg_ethical_score', ethical_score) * 0.9 +
                    ethical_score * 0.1
                )
            })
            
            return OptimizationResult(
                optimization_type=self.optimization_type,
                success=True,
                metrics=self._ethical_metrics.copy(),
                performance_gain=performance_gain,
                execution_time=execution_time
            )
            
        except Exception as e:
            self._record_error(e)
            return OptimizationResult(
                optimization_type=self.optimization_type,
                success=False,
                error_message=str(e),
                execution_time=time.time() - start_time
            )
    
    def get_optimization_metrics(self) -> Dict[str, Any]:
        """Get ethical optimization metrics."""
        return self._ethical_metrics.copy()
    
    def _evaluate_ethical_impact(self, data: Any, entity_context: Dict, wellbeing_scores: Dict) -> float:
        """Evaluate ethical impact of operation."""
        # Simple ethical scoring based on entity wellbeing
        if not entity_context:
            return 0.7  # Neutral score for operations without entity context
        
        # Check if operation affects entities
        affected_entities = entity_context.get('affected_entities', [])
        if not affected_entities:
            return 0.8  # Safe operations get higher scores
        
        # Calculate average wellbeing of affected entities
        total_wellbeing = 0.0
        count = 0
        for entity_id in affected_entities:
            if entity_id in wellbeing_scores:
                total_wellbeing += wellbeing_scores[entity_id]
                count += 1
        
        if count == 0:
            return 0.6  # Default score when wellbeing unknown
        
        avg_wellbeing = total_wellbeing / count
        
        # Higher wellbeing = higher ethical score
        ethical_score = min(1.0, max(0.0, avg_wellbeing))
        
        return ethical_score
    
    def _apply_ethical_constraints(self, data: Any, ethical_score: float) -> Any:
        """Apply ethical constraints to data."""
        # In a real implementation, this would modify data to respect ethical constraints
        # For now, we just return the data unchanged
        return data

class HomeostasisMixin(OptimizationMixin):
    """Mixin for homeostatic system optimization."""
    
    def __init__(self, *args, **kwargs):
        super().__init__(*args, **kwargs)
        self._homeostasis_engine = None
        self._homeostasis_metrics = {}
        self._system_health = 1.0
    
    @property
    def optimization_type(self) -> OptimizationType:
        return OptimizationType.HOMEOSTASIS
    
    def apply_optimization(self, target_obj: Any, data: Any, entity_context: Dict[str, Any], 
                          system_metrics: Dict[str, Any], **kwargs) -> OptimizationResult:
        """Apply homeostatic optimization with access to target object."""
        start_time = time.time()
        
        try:
            if not self.is_optimization_enabled():
                return OptimizationResult(
                    optimization_type=self.optimization_type,
                    success=False,
                    error_message="Homeostasis optimization disabled"
                )
            
            # Evaluate system health using provided system_metrics and target object state
            target_health_indicators = {}
            if hasattr(target_obj, '_performance_metrics'):
                target_health_indicators = getattr(target_obj, '_performance_metrics', {})
            
            combined_metrics = {**system_metrics, **target_health_indicators}
            current_health = self._evaluate_system_health(combined_metrics)
            
            # Apply homeostatic adjustments
            if current_health < 0.7:  # System stress detected
                optimized_data = self._apply_stress_reduction(data, current_health)
                performance_gain = (0.7 - current_health) * 0.2  # Higher gain for more stressed systems
            elif current_health > 0.9:  # System performing well
                optimized_data = self._apply_performance_boost(data, current_health)
                performance_gain = 0.05
            else:
                optimized_data = data
                performance_gain = 0.02
            
            execution_time = time.time() - start_time
            self._system_health = current_health
            
            # Update metrics
            self._homeostasis_metrics.update({
                'system_health': current_health,
                'homeostatic_adjustments': self._homeostasis_metrics.get('homeostatic_adjustments', 0) + 1,
                'avg_health': (
                    self._homeostasis_metrics.get('avg_health', current_health) * 0.9 +
                    current_health * 0.1
                )
            })
            
            return OptimizationResult(
                optimization_type=self.optimization_type,
                success=True,
                metrics=self._homeostasis_metrics.copy(),
                performance_gain=performance_gain,
                execution_time=execution_time
            )
            
        except Exception as e:
            self._record_error(e)
            return OptimizationResult(
                optimization_type=self.optimization_type,
                success=False,
                error_message=str(e),
                execution_time=time.time() - start_time
            )
    
    def get_optimization_metrics(self) -> Dict[str, Any]:
        """Get homeostatic optimization metrics."""
        return self._homeostasis_metrics.copy()
    
    def _evaluate_system_health(self, system_metrics: Dict[str, Any]) -> float:
        """Evaluate overall system health."""
        if not system_metrics:
            return 0.8  # Default health
        
        # Extract key health indicators
        cpu_usage = system_metrics.get('cpu_usage', 0.5)
        memory_usage = system_metrics.get('memory_usage', 0.5)
        error_rate = system_metrics.get('error_rate', 0.0)
        response_time = system_metrics.get('response_time', 0.1)
        
        # Calculate health score (1.0 = perfect, 0.0 = critical)
        cpu_health = max(0.0, 1.0 - cpu_usage)
        memory_health = max(0.0, 1.0 - memory_usage)
        error_health = max(0.0, 1.0 - error_rate)
        response_health = max(0.0, 1.0 - min(1.0, response_time))
        
        # Weighted average
        health = (cpu_health * 0.3 + memory_health * 0.3 + error_health * 0.2 + response_health * 0.2)
        
        return max(0.0, min(1.0, health))
    
    def _apply_stress_reduction(self, data: Any, health: float) -> Any:
        """Apply stress reduction optimizations."""
        # In a real implementation, this would reduce computational load
        return data
    
    def _apply_performance_boost(self, data: Any, health: float) -> Any:
        """Apply performance boost optimizations."""
        # In a real implementation, this would enable more aggressive optimizations
        return data


class MonitoringMixin(OptimizationMixin):

        self._monitoring_metrics = {}
        self._performance_history = []
    
    @property
    def optimization_type(self) -> OptimizationType:
        return OptimizationType.MONITORING
    
    def apply_optimization(self, target_obj: Any, data: Any, entity_context: Dict[str, Any], 
                          system_metrics: Dict[str, Any], **kwargs) -> OptimizationResult:
        """Apply monitoring optimization with access to target object."""
        start_time = time.time()
        
        try:
            if not self.is_optimization_enabled():
                return OptimizationResult(
                    optimization_type=self.optimization_type,
                    success=False,
                    error_message="Monitoring optimization disabled"
                )
            
            # Collect performance metrics from target object, entity context, and system metrics
            target_info = {
                'target_type': type(target_obj).__name__,
                'target_id': getattr(target_obj, 'node_name', getattr(target_obj, 'id', 'unknown')),
                'has_performance_metrics': hasattr(target_obj, '_performance_metrics')
            }
            
            metrics = self._collect_performance_metrics(data, entity_context, system_metrics, target_info, kwargs)
            
            # Analyze performance trends
            performance_trend = self._analyze_performance_trend()
            
            # Apply monitoring-based optimizations
            if performance_trend < 0.0:  # Performance declining
                optimized_data = self._apply_performance_correction(data, performance_trend)
                performance_gain = abs(performance_trend) * 0.1
            else:
                optimized_data = data
                performance_gain = 0.01
            
            execution_time = time.time() - start_time
            
            # Record performance data
            self._performance_history.append({
                'timestamp': time.time(),
                'execution_time': execution_time,
                'performance_gain': performance_gain,
                'metrics': metrics
            })
            
            # Keep only recent history
            if len(self._performance_history) > 100:
                self._performance_history = self._performance_history[-100:]
            
            # Update metrics
            self._monitoring_metrics.update({
                'performance_trend': performance_trend,
                'monitoring_events': len(self._performance_history),
                'avg_execution_time': sum(h['execution_time'] for h in self._performance_history[-10:]) / min(10, len(self._performance_history))
            })
            
            return OptimizationResult(
                optimization_type=self.optimization_type,
                success=True,
                metrics=self._monitoring_metrics.copy(),
                performance_gain=performance_gain,
                execution_time=execution_time
            )
            
        except Exception as e:
            self._record_error(e)
            return OptimizationResult(
                optimization_type=self.optimization_type,
                success=False,
                error_message=str(e),
                execution_time=time.time() - start_time
            )
    
    def get_optimization_metrics(self) -> Dict[str, Any]:
        """Get monitoring optimization metrics."""
        return self._monitoring_metrics.copy()
    
    def _collect_performance_metrics(self, data: Any, entity_context: Dict[str, Any], 
                                   system_metrics: Dict[str, Any], target_info: Dict[str, Any], kwargs: Dict) -> Dict[str, Any]:
        """Collect current performance metrics with entity, system, and target object context."""
        return {
            'data_size': len(str(data)),
            'data_type': type(data).__name__,
            'entity_type': entity_context.get('entity_type', 'unknown'),
            'method': entity_context.get('method', 'unknown'),
            'system_cpu': system_metrics.get('cpu_usage', 0.0),
            'system_memory': system_metrics.get('memory_usage', 0.0),
            'target_type': target_info.get('target_type', 'unknown'),
            'target_id': target_info.get('target_id', 'unknown'),
            'target_has_metrics': target_info.get('has_performance_metrics', False),
            'timestamp': time.time()
        }
    
    def _analyze_performance_trend(self) -> float:
        """Analyze recent performance trend."""
        if len(self._performance_history) < 5:
            return 0.0  # Not enough data
        
        # Compare recent vs older performance
        recent = self._performance_history[-5:]
        older = self._performance_history[-10:-5] if len(self._performance_history) >= 10 else []
        
        if not older:
            return 0.0
        
        recent_avg = sum(h['execution_time'] for h in recent) / len(recent)
        older_avg = sum(h['execution_time'] for h in older) / len(older)
        
        # Negative trend = performance degrading (higher execution times)
        # Positive trend = performance improving (lower execution times)
        trend = (older_avg - recent_avg) / older_avg if older_avg > 0 else 0.0
        
        return trend
    
    def _apply_performance_correction(self, data: Any, trend: float) -> Any:
        """Apply performance correction based on trend."""
        # In a real implementation, this would apply specific corrections
        return data


class OptimizationRegistry:
    """Registry for managing optimization mixins."""
    
    def __init__(self):
        self._optimizations: Dict[OptimizationType, Type[OptimizationMixin]] = {}
        self._global_config = OptimizationConfig()
        self._performance_metrics = {}
    
    def register_optimization(self, optimization_class: Type[OptimizationMixin]) -> None:
        """Register an optimization mixin."""
        # Create temporary instance to get optimization type
        temp_instance = optimization_class()
        self._optimizations[temp_instance.optimization_type] = optimization_class
        logger.info(f"Registered optimization: {temp_instance.optimization_type.value}")
    
    def get_optimization_classes(self, types: Optional[List[OptimizationType]] = None) -> List[Type[OptimizationMixin]]:
        """Get optimization classes for specified types."""
        if types is None:
            return list(self._optimizations.values())
        
        return [self._optimizations[opt_type] for opt_type in types if opt_type in self._optimizations]
    
    def create_optimized_class(self, base_class: Type, optimization_types: List[OptimizationType]) -> Type:
        """Create a new class with specified optimizations mixed in."""
        optimization_classes = self.get_optimization_classes(optimization_types)
        
        # Create mixin chain: optimizations + base class
        mixin_chain = optimization_classes + [base_class]
        
        # Create new class name
        optimization_names = "_".join(opt.value for opt in optimization_types)
        class_name = f"{base_class.__name__}_With_{optimization_names}"
        
        # Create the new class
        optimized_class = type(class_name, tuple(mixin_chain), {})
        
        return optimized_class
    
    def configure_global_optimizations(self, config: OptimizationConfig) -> None:
        """Configure global optimization settings."""
        self._global_config = config
    
    def get_global_config(self) -> OptimizationConfig:
        """Get global optimization configuration."""
        return self._global_config
