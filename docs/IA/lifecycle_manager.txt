"""
LifecycleManager - Gestor de Ciclo de Vida de Entidades Vivientes
=================================================================
Gestiona el ciclo de vida de LivingEntity en la simulación, incluyendo
agregado, suspensión, reactivación, actualización física y manejo de acciones pendientes.
Optimizado para concurrencia y eficiencia en recursos.
Extensión EVA: integra memoria viviente, ingestión/recall de experiencias de ciclo de vida,
faseo, hooks de entorno y benchmarking.
"""

import logging
import multiprocessing
from typing import Any

from crisalida_lib.ADAM.adam_core import LivingEntity
from crisalida_lib.network.noosphere_client import NoosphereClient
from crisalida_lib.symbolic_language.living_symbols import LivingSymbolRuntime
from crisalida_lib.symbolic_language.types import (
    EVAExperience,
    QualiaState,
    RealityBytecode,
)

from ..network.p2p_dht_network import JanusDHTNode

logger = logging.getLogger(__name__)


def _update_entity_worker(
    entity_state: bytes,
    delta_time: float,
    simulation_tick: int,
    unified_field_proxy: dict[str, Any],
    simulation_lattice_proxy: dict[str, Any],
    dht_node_info: dict[str, Any],
    simulation_entities_proxy: dict[str, Any],
) -> tuple[
    bytes,
    float,
    dict[str, Any],
    dict[str, Any],
    dict[str, Any],
    list[dict[str, Any]],
]:
    entity = LivingEntity.from_updatable_state(entity_state)
    JanusDHTNode(
        host=dht_node_info["host"],
        port=dht_node_info["port"],
        node_id=dht_node_info["node_id"],
    )
    noosphere_client = NoosphereClient(
        dht_node_info=dht_node_info, entity_id=entity.entity_id
    )
    entity.update_internal_state(
        delta_time,
        simulation_tick,
        unified_field_proxy,
        simulation_lattice_proxy,
        noosphere_client,
    )
    return (
        entity.to_updatable_state(),
        entity.qualia_state.consciousness_density,
        unified_field_proxy,
        simulation_lattice_proxy,
        simulation_entities_proxy,
        entity.pending_actions,
    )


class LifecycleManager:
    """
    Gestor central del ciclo de vida de entidades vivientes en la simulación.
    Extensión EVA: memoria viviente, ingestión/recall de experiencias de ciclo de vida,
    faseo, hooks de entorno y benchmarking.
    """

    def __init__(self, physics_engine, eva_phase: str = "default"):
        self.entities: dict[str, LivingEntity] = {}
        self.physics_engine = physics_engine
        # EVA extension
        self.eva_phase = eva_phase
        self.eva_runtime = LivingSymbolRuntime()
        self.eva_memory_store: dict[str, RealityBytecode] = {}
        self.eva_experience_store: dict[str, EVAExperience] = {}
        self.eva_phases: dict[str, dict[str, RealityBytecode]] = {}
        self._environment_hooks: list = []

    def add_entity(self, entity: LivingEntity):
        self.entities[entity.entity_id] = entity
        logger.info(f"[LifecycleManager] Entity {entity.entity_id} added.")
        # EVA: Ingesta experiencia de ciclo de vida (agregado)
        self.eva_ingest_lifecycle_experience(
            entity.entity_id,
            event_type="added",
            details={"entity": entity.entity_id},
            qualia_state=getattr(entity, "qualia_state", None),
            phase=self.eva_phase,
        )

    def remove_entity(self, entity_id: str):
        if entity_id in self.entities:
            del self.entities[entity_id]
            logger.info(f"[LifecycleManager] Entity {entity_id} removed.")
            # EVA: Ingesta experiencia de ciclo de vida (removido)
            self.eva_ingest_lifecycle_experience(
                entity_id,
                event_type="removed",
                details={"entity": entity_id},
                qualia_state=None,
                phase=self.eva_phase,
            )

    def get_total_resource_consumption(self) -> float:
        return sum(
            entity.resource_consumption_rate
            for entity in self.entities.values()
            if not getattr(entity, "is_suspended", False)
        )

    def manage_entity_lifecycle(self, max_resource_limit: float):
        current_consumption = self.get_total_resource_consumption()
        if current_consumption > max_resource_limit:
            sorted_entities = sorted(
                [
                    e
                    for e in self.entities.values()
                    if not getattr(e, "is_suspended", False)
                ],
                key=lambda e: (
                    e.qualia_state.consciousness_density,
                    -e.resource_consumption_rate,
                ),
            )
            for entity in sorted_entities:
                if current_consumption > max_resource_limit:
                    if not getattr(entity, "is_suspended", False):
                        entity.suspend()
                        current_consumption -= entity.resource_consumption_rate
                        # EVA: Ingesta experiencia de ciclo de vida (suspendido)
                        self.eva_ingest_lifecycle_experience(
                            entity.entity_id,
                            event_type="suspended",
                            details={"entity": entity.entity_id},
                            qualia_state=getattr(entity, "qualia_state", None),
                            phase=self.eva_phase,
                        )
                else:
                    break
        else:
            sorted_suspended_entities = sorted(
                [
                    e
                    for e in self.entities.values()
                    if getattr(e, "is_suspended", False)
                ],
                key=lambda e: e.qualia_state.consciousness_density,
                reverse=True,
            )
            for entity in sorted_suspended_entities:
                if (
                    current_consumption + entity.resource_consumption_rate
                    <= max_resource_limit
                ):
                    entity.resume()
                    current_consumption += entity.resource_consumption_rate
                    # EVA: Ingesta experiencia de ciclo de vida (reactivado)
                    self.eva_ingest_lifecycle_experience(
                        entity.entity_id,
                        event_type="resumed",
                        details={"entity": entity.entity_id},
                        qualia_state=getattr(entity, "qualia_state", None),
                        phase=self.eva_phase,
                    )

    def _prepare_worker_args(
        self, entities_to_update: list[LivingEntity], simulation_tick: int
    ) -> list[tuple]:
        unified_field_proxy: dict[str, Any] = {
            "current_reality_resource": self.physics_engine.simulation_instance.unified_field.current_reality_resource
        }
        simulation_lattice_proxy: dict[str, Any] = {"actions": []}
        noosphere_client_proxy: dict[str, Any] = (
            self.physics_engine.simulation_instance.p2p_manager.dht_node.node_info.to_dict()
        )
        simulation_entities_proxy: dict[str, Any] = {}
        worker_args = [
            (
                entity.to_updatable_state(),
                1.0,
                simulation_tick,
                unified_field_proxy,
                simulation_lattice_proxy,
                noosphere_client_proxy,
                simulation_entities_proxy,
            )
            for entity in entities_to_update
        ]
        return worker_args

    def _process_worker_results(
        self,
        updated_results: list[tuple],
        dead_entities: list[str],
        current_lattice: Any,
    ) -> list[dict[str, Any]]:
        all_pending_actions = []
        for updated_result in updated_results:
            updated_entity_state = updated_result[0]
            _ = updated_result[1]
            returned_unified_field_proxy = updated_result[2]
            returned_simulation_lattice_proxy = updated_result[3]
            returned_simulation_entities_proxy = updated_result[4]
            pending_actions = updated_result[5]
            updated_entity = LivingEntity.from_updatable_state(updated_entity_state)
            self.entities[updated_entity.entity_id] = updated_entity
            self.physics_engine.simulation_instance.unified_field.current_reality_resource = returned_unified_field_proxy[
                "current_reality_resource"
            ]
            for action in returned_simulation_lattice_proxy.get("actions", []):
                if action["type"] == "apply_force":
                    self.physics_engine.simulation_instance.simulation_lattice.apply_force_to_lattice(
                        action["lattice_type"],
                        self.physics_engine.simulation_instance.simulation_lattice.world_to_lattice_coords(
                            action["position"]
                        ),
                        action["force_magnitude"],
                    )
            for entity_id, status in returned_simulation_entities_proxy.items():
                if status == "REMOVE":
                    if entity_id not in dead_entities:
                        dead_entities.append(entity_id)
            all_pending_actions.extend(pending_actions)
            self.physics_engine.apply_physics(updated_entity, current_lattice)
            if not updated_entity.is_alive():
                if updated_entity.entity_id not in dead_entities:
                    dead_entities.append(updated_entity.entity_id)
                    # EVA: Ingesta experiencia de ciclo de vida (muerto)
                    self.eva_ingest_lifecycle_experience(
                        updated_entity.entity_id,
                        event_type="dead",
                        details={"entity": updated_entity.entity_id},
                        qualia_state=getattr(updated_entity, "qualia_state", None),
                        phase=self.eva_phase,
                    )
        return all_pending_actions

    def _process_pending_actions(self, all_pending_actions: list[dict[str, Any]]):
        for action in all_pending_actions:
            if action["type"] == "publish_post":
                main_storage_manager = (
                    self.physics_engine.simulation_instance.storage_manager
                )
                main_noosphere_client = NoosphereClient(
                    dht_node_info=main_storage_manager.p2p_node.node_info.to_dict(),
                    entity_id=action["author_entity_id"],
                )
                import asyncio

                asyncio.run(
                    main_noosphere_client.publish_post(
                        content_text=action["content_text"],
                        post_type=action["post_type"],
                        qualia_state_json=action["qualia_state_json"],
                    )
                )

    def _handle_updated_entities_and_actions(
        self,
        updated_results: list[tuple],
        dead_entities: list[str],
        current_lattice: Any,
    ):
        all_pending_actions = self._process_worker_results(
            updated_results, dead_entities, current_lattice
        )
        self._process_pending_actions(all_pending_actions)

    def update_all(self, ticks: int, current_lattice: Any = None):
        dead_entities: list[str] = []
        entities_to_update = [
            entity
            for entity in self.entities.values()
            if not getattr(entity, "is_suspended", False) and entity.is_alive()
        ]
        simulation_tick = getattr(
            self.physics_engine.simulation_instance, "simulation_tick", 0
        )
        worker_args = self._prepare_worker_args(entities_to_update, simulation_tick)
        with multiprocessing.Pool() as pool:
            updated_results = pool.starmap(_update_entity_worker, worker_args)
        self._handle_updated_entities_and_actions(
            updated_results, dead_entities, current_lattice
        )
        for entity_id in dead_entities:
            self.remove_entity(entity_id)

    # === EVA EXTENSION ===

    def eva_ingest_lifecycle_experience(
        self,
        entity_id: str,
        event_type: str,
        details: dict = None,
        qualia_state: QualiaState = None,
        phase: str = None,
    ) -> str:
        """
        Compila una experiencia de ciclo de vida en RealityBytecode y la almacena en la memoria EVA.
        """
        import time

        phase = phase or self.eva_phase
        qualia_state = qualia_state or QualiaState(
            emotional_valence=0.5,
            cognitive_complexity=0.7,
            consciousness_density=0.6,
            narrative_importance=1.0,
            energy_level=1.0,
        )
        experience_id = f"eva_lifecycle_{entity_id}_{event_type}_{int(time.time())}"
        experience_data = {
            "entity_id": entity_id,
            "event_type": event_type,
            "details": details or {},
            "qualia_state": (
                qualia_state.to_dict() if hasattr(qualia_state, "to_dict") else {}
            ),
            "timestamp": time.time(),
            "phase": phase,
        }
        intention = {
            "intention_type": "ARCHIVE_LIFECYCLE_EXPERIENCE",
            "experience": experience_data,
            "qualia": qualia_state,
            "phase": phase,
        }
        bytecode = self.eva_runtime.divine_compiler.compile_intention(intention)
        reality_bytecode = RealityBytecode(
            bytecode_id=experience_id,
            instructions=bytecode,
            qualia_state=qualia_state,
            phase=phase,
            timestamp=experience_data["timestamp"],
        )
        self.eva_memory_store[experience_id] = reality_bytecode
        if phase not in self.eva_phases:
            self.eva_phases[phase] = {}
        self.eva_phases[phase][experience_id] = reality_bytecode
        self.eva_experience_store[experience_id] = reality_bytecode
        for hook in self._environment_hooks:
            try:
                hook(reality_bytecode)
            except Exception as e:
                logger.warning(f"[EVA-LIFECYCLE] Environment hook failed: {e}")
        return experience_id

    def eva_recall_lifecycle_experience(self, cue: str, phase: str = None) -> dict:
        """
        Ejecuta el RealityBytecode de una experiencia de ciclo de vida almacenada, manifestando la simulación.
        """
        phase = phase or self.eva_phase
        reality_bytecode = self.eva_phases.get(phase, {}).get(
            cue
        ) or self.eva_memory_store.get(cue)
        if not reality_bytecode:
            return {"error": "No bytecode found for EVA lifecycle experience"}
        quantum_field = getattr(self.eva_runtime, "quantum_field", None)
        manifestations = []
        if quantum_field:
            for instr in reality_bytecode.instructions:
                symbol_manifest = self.eva_runtime.execute_instruction(
                    instr, quantum_field
                )
                if symbol_manifest:
                    manifestations.append(symbol_manifest)
                    for hook in self._environment_hooks:
                        try:
                            hook(symbol_manifest)
                        except Exception as e:
                            logger.warning(
                                f"[EVA-LIFECYCLE] Manifestation hook failed: {e}"
                            )
        eva_experience = EVAExperience(
            experience_id=reality_bytecode.bytecode_id,
            bytecode=reality_bytecode,
            manifestations=manifestations,
            phase=reality_bytecode.phase,
            qualia_state=reality_bytecode.qualia_state,
            timestamp=reality_bytecode.timestamp,
        )
        self.eva_experience_store[reality_bytecode.bytecode_id] = eva_experience
        return {
            "experience_id": eva_experience.experience_id,
            "manifestations": [m.to_dict() for m in manifestations],
            "phase": eva_experience.phase,
            "qualia_state": (
                eva_experience.qualia_state.to_dict()
                if hasattr(eva_experience.qualia_state, "to_dict")
                else {}
            ),
            "timestamp": eva_experience.timestamp,
        }

    def add_experience_phase(
        self,
        experience_id: str,
        phase: str,
        entity_id: str,
        event_type: str,
        details: dict = None,
        qualia_state: QualiaState = None,
    ):
        """
        Añade una fase alternativa para una experiencia de ciclo de vida EVA.
        """
        import time

        qualia_state = qualia_state or QualiaState(
            emotional_valence=0.5,
            cognitive_complexity=0.7,
            consciousness_density=0.6,
            narrative_importance=1.0,
            energy_level=1.0,
        )
        experience_data = {
            "entity_id": entity_id,
            "event_type": event_type,
            "details": details or {},
            "qualia_state": (
                qualia_state.to_dict() if hasattr(qualia_state, "to_dict") else {}
            ),
            "timestamp": time.time(),
            "phase": phase,
        }
        intention = {
            "intention_type": "ARCHIVE_LIFECYCLE_EXPERIENCE",
            "experience": experience_data,
            "qualia": qualia_state,
            "phase": phase,
        }
        bytecode = self.eva_runtime.divine_compiler.compile_intention(intention)
        reality_bytecode = RealityBytecode(
            bytecode_id=experience_id,
            instructions=bytecode,
            qualia_state=qualia_state,
            phase=phase,
            timestamp=experience_data["timestamp"],
        )
        if phase not in self.eva_phases:
            self.eva_phases[phase] = {}
        self.eva_phases[phase][experience_id] = reality_bytecode

    def set_memory_phase(self, phase: str):
        """Cambia la fase activa de memoria EVA."""
        self.eva_phase = phase
        for hook in self._environment_hooks:
            try:
                hook({"phase_changed": phase})
            except Exception as e:
                logger.warning(f"[EVA-LIFECYCLE] Phase hook failed: {e}")

    def get_memory_phase(self) -> str:
        """Devuelve la fase de memoria actual."""
        return self.eva_phase

    def get_experience_phases(self, experience_id: str) -> list:
        """Lista todas las fases disponibles para una experiencia de ciclo de vida EVA."""
        return [
            phase for phase, exps in self.eva_phases.items() if experience_id in exps
        ]

    def add_environment_hook(self, hook: callable):
        """Registra un hook para manifestación simbólica o eventos EVA."""
        self._environment_hooks.append(hook)

    def get_eva_api(self) -> dict:
        return {
            "eva_ingest_lifecycle_experience": self.eva_ingest_lifecycle_experience,
            "eva_recall_lifecycle_experience": self.eva_recall_lifecycle_experience,
            "add_experience_phase": self.add_experience_phase,
            "set_memory_phase": self.set_memory_phase,
            "get_memory_phase": self.get_memory_phase,
            "get_experience_phases": self.get_experience_phases,
            "add_environment_hook": self.add_environment_hook,
        }
