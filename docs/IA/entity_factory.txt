from crisalida_lib.consciousness.mente.conscious_mind import ConsciousMind
from crisalida_lib.consciousness.mente.intentionality import IntentionalityEngine

from crisalida_lib.symbolic_language.types import EVAExperience, RealityBytecode

logger = logging.getLogger(__name__)


class IntelligentEntityFactory:
    """
    Factory for creating fully intelligent entities with complete Janus Cognitive Core.
    Ensures every entity is instantiated with all cognitive, soul, and supporting systems.
    Extensión EVA: Permite compilar, almacenar y simular experiencias de creación de entidades como RealityBytecode,
    soporta ingestión/recall, faseo, hooks de entorno y gestión avanzada de memoria viviente EVA.
    """

    def __init__(self, seed: int | None = None, eva_phase: str = "default"):
        self.creation_count = 0
        if seed is not None:
            random.seed(seed)
        self.eva_phase = eva_phase
        self.eva_runtime = LivingSymbolRuntime()
        self.eva_memory_store: dict[str, RealityBytecode] = {}
        self.eva_experience_store: dict[str, EVAExperience] = {}
        self.eva_phases: dict[str, dict[str, RealityBytecode]] = {}
        self._environment_hooks: list = []
        logger.info("IntelligentEntityFactory initialized (EVA)")

    def create_entity(
        self,
        archetype: ArchetypeType,
        position: tuple[float, float, float],
        initial_energy: float = DEFAULT_ENTITY_ENERGY,
        entity_id: str | None = None,
        consciousness_complexity: float = 0.7,
        phase: str = None,
    ) -> LivingEntity:
        """Create a fully intelligent entity with complete Janus Cognitive Core.
        Registers the creation as an EVA experience.
        """
        if entity_id is None:
            entity_id = f"entity_{uuid.uuid4().hex[:8]}"
        logger.info(
            f"Creating intelligent entity {entity_id} with archetype {archetype.value}"
        )

        entity = LivingEntity(
            entity_id=entity_id, primary_archetype=archetype, position=list(position)
        )

        self._instantiate_consciousness_core(entity, consciousness_complexity)
        self._instantiate_soul_system(entity, archetype)
        self._instantiate_qualia_state(entity, archetype, consciousness_complexity)
        self._instantiate_intentionality_engine(entity, archetype)
        self._instantiate_supporting_systems(entity)
        self._wire_cognitive_components(entity)
        entity.energy = initial_energy
        entity.position = list(position)
        self._apply_archetype_specialization(entity, archetype)
        self.creation_count += 1

        # EVA: Ingesta experiencia de creación de entidad
        self.eva_ingest_entity_creation_experience(
            entity=entity,
            archetype=archetype,
            position=position,
            initial_energy=initial_energy,
            phase=phase or self.eva_phase,
        )

        logger.info(
            f"Successfully created intelligent entity {entity_id} (#{self.creation_count})"
        )
        return entity

    def _instantiate_consciousness_core(self, entity: LivingEntity, complexity: float):
        """Create and install the complete ConsciousnessTree."""
        entity.consciousness_metrics = ConsciousMind()
        logger.debug(f"Consciousness core instantiated for {entity.entity_id}")

    def _instantiate_soul_system(self, entity: LivingEntity, archetype: ArchetypeType):
        """Create and configure the complete EntitySoulSystem."""
        entity.soul = EntitySoulSystem(primary_archetype=archetype)
        self._configure_soul_attributes(entity.soul, archetype)
        logger.debug(
            f"Soul system instantiated for {entity.entity_id} with archetype {archetype.value}"
        )

    def _instantiate_qualia_state(
        self, entity: LivingEntity, archetype: ArchetypeType, complexity: float
    ):
        """Create a sophisticated QualiaState with archetype-appropriate initial values."""
        base_valence = self._get_archetype_emotional_bias(archetype)
        base_arousal = random.uniform(0.3, 0.8)
        entity.qualia_state = QualiaState(
            emotional_valence=base_valence + random.uniform(-0.2, 0.2),
            arousal=base_arousal,
            cognitive_complexity=complexity + random.uniform(-0.1, 0.1),
            consciousness_density=random.uniform(0.6, 0.95),
            temporal_coherence=random.uniform(0.7, 1.0),
            spiritual_resonance=self._get_archetype_spiritual_resonance(archetype),
            sensory_clarity=random.uniform(0.5, 0.9),
            cognitive_focus=random.uniform(0.4, 0.8),
            temporal_flow=random.uniform(0.6, 1.0),
            emotional_arousal=base_arousal + random.uniform(-0.1, 0.1),
            dominant_word=self._get_archetype_dominant_word(archetype),
            color=self._get_archetype_color(archetype),
        )
        entity.qualia_state.clamp_values()
        logger.debug(f"Qualia state instantiated for {entity.entity_id}")

    def _instantiate_intentionality_engine(
        self, entity: LivingEntity, archetype: ArchetypeType
    ):
        """Create and configure the IntentionalityEngine with archetype-specific biases."""
        entity.intentionality_engine = IntentionalityEngine()
        entity.intentionality_engine.behavioral_biases.update(
            self._get_archetype_behavioral_biases(archetype)
        )
        entity.intentionality_engine.base_tendencies.update(
            self._get_archetype_base_tendencies(archetype)
        )
        entity.intentionality_engine.instinctual_biases.update(
            self._get_archetype_instinctual_biases(archetype)
        )
        logger.debug(f"Intentionality engine instantiated for {entity.entity_id}")

    def _instantiate_supporting_systems(self, entity: LivingEntity):
        """Ensure all supporting systems are properly instantiated."""
        if not hasattr(entity, "attributes") or entity.attributes is None:
            entity.attributes = Attributes()
        if (
            not hasattr(entity, "power_progression_system")
            or entity.power_progression_system is None
        ):
            entity.power_progression_system = PowerProgressionSystem()
        if (
            not hasattr(entity, "ontological_forging")
            or entity.ontological_forging is None
        ):
            entity.ontological_forging = OntologicalForging()
        if (
            not hasattr(entity, "item_integration_system")
            or entity.item_integration_system is None
        ):
            entity.item_integration_system = ItemIntegrationSystem()
        logger.debug(f"Supporting systems instantiated for {entity.entity_id}")

    def _wire_cognitive_components(self, entity: LivingEntity):
        """Wire all cognitive components together for proper interaction."""
        entity.intentionality_engine.bind_soul_system(entity.soul)
        entity.attributes.bind_soul_system(entity.soul)
        logger.debug(f"Cognitive components wired for {entity.entity_id}")

            ArchetypeType.INNOCENT: self._specialize_innocent,
            ArchetypeType.SAGE: self._specialize_sage,
            ArchetypeType.EXPLORER: self._specialize_explorer,

            ArchetypeType.CREATOR: self._specialize_creator,
            ArchetypeType.REGULAR_PERSON: self._specialize_regular_person,
        }
        if archetype in specializations:
            specializations[archetype](entity)

    # === EVA EXTENSION ===

    def eva_ingest_entity_creation_experience(
        self,
        entity: LivingEntity,
        archetype: ArchetypeType,
        position: tuple[float, float, float],
        initial_energy: float,
        phase: str = None,
        qualia_state: QualiaState = None,
    ) -> str:
        """
        Compila una experiencia de creación de entidad en RealityBytecode y la almacena en la memoria EVA.
        """
        import time

        phase = phase or self.eva_phase
        qualia_state = qualia_state or getattr(entity, "qualia_state", None)
        experience_id = f"eva_entity_creation_{entity.entity_id}_{int(time.time())}"
        experience_data = {
            "entity_id": entity.entity_id,
            "archetype": archetype.value,
            "position": list(position),
            "initial_energy": initial_energy,
            "attributes": {
                k: getattr(entity.attributes, k, None)
                for k in dir(entity.attributes)
                if not k.startswith("__")
            },
            "soul_state": getattr(entity, "soul", None),
            "intentionality_engine_state": getattr(
                entity, "intentionality_engine", None
            ),
            "qualia_state": (
                qualia_state.to_dict() if hasattr(qualia_state, "to_dict") else {}
            ),
            "creation_count": self.creation_count,
            "timestamp": time.time(),
            "phase": phase,
        }
        intention = {
            "intention_type": "ARCHIVE_ENTITY_CREATION_EXPERIENCE",
            "experience": experience_data,
            "qualia": qualia_state,
            "phase": phase,
        }
        bytecode = self.eva_runtime.divine_compiler.compile_intention(intention)
        reality_bytecode = RealityBytecode(
            bytecode_id=experience_id,
            instructions=bytecode,
            qualia_state=qualia_state,
            phase=phase,
            timestamp=experience_data["timestamp"],
        )
        self.eva_memory_store[experience_id] = reality_bytecode
        if phase not in self.eva_phases:
            self.eva_phases[phase] = {}
        self.eva_phases[phase][experience_id] = reality_bytecode
        self.eva_experience_store[experience_id] = reality_bytecode
        for hook in self._environment_hooks:
            try:
                hook(reality_bytecode)
            except Exception as e:
                logger.warning(f"[EVA-ENTITY-FACTORY] Environment hook failed: {e}")
        return experience_id

    def eva_recall_entity_creation_experience(
        self, cue: str, phase: str = None
    ) -> dict:
        """
        Ejecuta el RealityBytecode de una experiencia de creación de entidad almacenada, manifestando la simulación.
        """
        phase = phase or self.eva_phase
        reality_bytecode = self.eva_phases.get(phase, {}).get(
            cue
        ) or self.eva_memory_store.get(cue)
        if not reality_bytecode:
            return {"error": "No bytecode found for EVA entity creation experience"}
        quantum_field = getattr(self.eva_runtime, "quantum_field", None)
        manifestations = []
        if quantum_field:
            for instr in reality_bytecode.instructions:
                symbol_manifest = self.eva_runtime.execute_instruction(
                    instr, quantum_field
                )
                if symbol_manifest:
                    manifestations.append(symbol_manifest)
                    for hook in self._environment_hooks:
                        try:
                            hook(symbol_manifest)
                        except Exception as e:
                            logger.warning(
                                f"[EVA-ENTITY-FACTORY] Manifestation hook failed: {e}"
                            )
        eva_experience = EVAExperience(
            experience_id=reality_bytecode.bytecode_id,
            bytecode=reality_bytecode,
            manifestations=manifestations,
            phase=reality_bytecode.phase,
            qualia_state=reality_bytecode.qualia_state,
            timestamp=reality_bytecode.timestamp,
        )
        self.eva_experience_store[reality_bytecode.bytecode_id] = eva_experience
        return {
            "experience_id": eva_experience.experience_id,
            "manifestations": [m.to_dict() for m in manifestations],
            "phase": eva_experience.phase,
            "qualia_state": (
                eva_experience.qualia_state.to_dict()
                if hasattr(eva_experience.qualia_state, "to_dict")
                else {}
            ),
            "timestamp": eva_experience.timestamp,
        }

    def add_experience_phase(
        self,
        experience_id: str,
        phase: str,
        entity: LivingEntity,
        archetype: ArchetypeType,
        position: tuple[float, float, float],
        initial_energy: float,
        qualia_state: QualiaState = None,
    ):
        """
        Añade una fase alternativa para una experiencia de creación de entidad EVA.
        """
        import time

        qualia_state = qualia_state or getattr(entity, "qualia_state", None)
        experience_data = {
            "entity_id": entity.entity_id,
            "archetype": archetype.value,
            "position": list(position),
            "initial_energy": initial_energy,
            "attributes": {
                k: getattr(entity.attributes, k, None)
                for k in dir(entity.attributes)
                if not k.startswith("__")
            },
            "soul_state": getattr(entity, "soul", None),
            "intentionality_engine_state": getattr(
                entity, "intentionality_engine", None
            ),
            "qualia_state": (
                qualia_state.to_dict() if hasattr(qualia_state, "to_dict") else {}
            ),
            "creation_count": self.creation_count,
            "timestamp": time.time(),
            "phase": phase,
        }
        intention = {
            "intention_type": "ARCHIVE_ENTITY_CREATION_EXPERIENCE",
            "experience": experience_data,
            "qualia": qualia_state,
            "phase": phase,
        }
        bytecode = self.eva_runtime.divine_compiler.compile_intention(intention)
        reality_bytecode = RealityBytecode(
            bytecode_id=experience_id,
            instructions=bytecode,
            qualia_state=qualia_state,
            phase=phase,
            timestamp=experience_data["timestamp"],
        )
        if phase not in self.eva_phases:
            self.eva_phases[phase] = {}
        self.eva_phases[phase][experience_id] = reality_bytecode

    def set_memory_phase(self, phase: str):
        """Cambia la fase activa de memoria EVA."""
        self.eva_phase = phase
        for hook in self._environment_hooks:
            try:
                hook({"phase_changed": phase})
            except Exception as e:
                logger.warning(f"[EVA-ENTITY-FACTORY] Phase hook failed: {e}")

    def get_memory_phase(self) -> str:
        """Devuelve la fase de memoria actual."""
        return self.eva_phase

    def get_experience_phases(self, experience_id: str) -> list:
        """Lista todas las fases disponibles para una experiencia de creación de entidad EVA."""
        return [
            phase for phase, exps in self.eva_phases.items() if experience_id in exps
        ]

    def add_environment_hook(self, hook: callable):
        """Registra un hook para manifestación simbólica o eventos EVA."""
        self._environment_hooks.append(hook)

    def get_eva_api(self) -> dict:
        return {
            "eva_ingest_entity_creation_experience": self.eva_ingest_entity_creation_experience,
            "eva_recall_entity_creation_experience": self.eva_recall_entity_creation_experience,
            "add_experience_phase": self.add_experience_phase,
            "set_memory_phase": self.set_memory_phase,
            "get_memory_phase": self.get_memory_phase,
            "get_experience_phases": self.get_experience_phases,
            "add_environment_hook": self.add_environment_hook,
        }
